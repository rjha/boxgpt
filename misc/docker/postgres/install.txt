# READ ONLY USER 
CREATE USER xapi_worker WITH PASSWORD 'worker_secure_password';

-- 2. Allow the worker to establish a connection to the database
GRANT CONNECT ON DATABASE my_vector_db TO xapi_worker;

-- 3. Open the public schema door and grant access to existing tables
GRANT USAGE ON SCHEMA public TO xapi_worker;
GRANT SELECT ON ALL TABLES IN SCHEMA public TO xapi_worker;
GRANT SELECT ON ALL SEQUENCES IN SCHEMA public TO xapi_worker;

-- 4. Secure future tables created by the active session or xapi_admin
ALTER DEFAULT PRIVILEGES IN SCHEMA public 
    GRANT SELECT ON TABLES TO xapi_worker;
ALTER DEFAULT PRIVILEGES FOR ROLE xapi_admin IN SCHEMA public 
    GRANT SELECT ON TABLES TO xapi_worker;
